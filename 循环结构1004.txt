#include <stdio.h>
main(){
    int i1,i,j,T,n,mid;
    scanf("%d\n",&T);
    while(T--)
    {
    scanf("%d",&n);
    mid=n/2;
    for(i1=0;i1<=mid;i1++){
        for(j=0;j<mid-i1;j++){
            printf(" ");
        }
        for(j=0;j<i1*2+1;j++){
            printf("*");
        }        
        printf("\n");
    }
    for(i=mid-1;i>=0;i--){
        for(j=0;j<mid-i;j++){
            printf(" ");
        }
        for(j=0;j<2*i+1;j++){
            printf("*");
        }        
        printf("\n");
    }
    }
    return 0;
}
