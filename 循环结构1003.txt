#include <stdio.h>
main(){
    char ch;
    int i,j,n;
    scanf("%c",&ch);
    n=ch-64;
    for(i=0;i<n;i++){
        for(j=1;j<=n-i-1;j++){
            printf(" ");
        }
        for(j=0;j<=i;j++){
            printf("%c",'A'+j);
        }
        for(j=i-1;j>=0;j--){
            printf("%c",'A'+j);
        }
        printf("\n");
    }
    return 0;
}
